```tsx
export default function PreviewPdfPage() {
  const [sinal, setSinal] = useState(true)
  const [status, setStatus] = useState<'rascunho' | 'enviado' | 'aprovado'>('rascunho')
  const enviado = status === 'enviado' || status === 'aprovado'

  return (
    <AppLayout active="orcamentos">

      {/* BARRA DE AÇÕES */}
      <div style={{
        background: '#fff', borderBottom: '1px solid #EFEDE8',
        padding: '14px 0', marginBottom: 0,
        marginLeft: -40, marginRight: -40, marginTop: -34,
        paddingLeft: 40, paddingRight: 40,
      }}>
        <div style={{ maxWidth: 820, margin: '0 auto', display: 'flex', alignItems: 'center', justifyContent: 'space-between', gap: 16, flexWrap: 'wrap' }}>
          <div style={{ minWidth: 0 }}>
            {/* Breadcrumb */}
            <div style={{ display: 'flex', alignItems: 'center', gap: 7, fontSize: 12.5, color: '#A29E96', marginBottom: 6 }}>
              <a href="/orcamentos" style={{ color: '#A29E96', textDecoration: 'none', fontWeight: 500 }}
                onMouseEnter={e => e.currentTarget.style.color = '#2A9D8F'}
                onMouseLeave={e => e.currentTarget.style.color = '#A29E96'}
              >
                Orçamentos
              </a>
              <Icons.chevron style={{ color: '#CFCBC3' }} />
              <span style={{ color: '#5C594F', fontWeight: 600, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>
                #0042 — Mariana Costa
              </span>
            </div>
            <div style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
              <h1 style={{ margin: 0, fontSize: 20, fontWeight: 700, letterSpacing: '-0.02em', color: '#3A372F' }}>
                Preview do orçamento
              </h1>
              <StatusBadge status={status === 'rascunho' ? 'Rascunho' : status === 'enviado' ? 'Enviado' : 'Aprovado'} size="sm" />
            </div>
          </div>

          {/* Botões de ação */}
          <div style={{ display: 'flex', alignItems: 'center', gap: 10, flexWrap: 'wrap' }}>
            {!enviado && (
              <Button variant="ghost" icon={<Icons.back />} onClick={() => {}}>
                Editar orçamento
              </Button>
            )}
            <Button variant="secondary" icon={<Icons.download />} onClick={() => {}}>
              Baixar PDF
            </Button>
            {!enviado ? (
              <Button variant="primary" icon={<Icons.send />} iconRight={<Icons.arrowRight />} onClick={() => setStatus('enviado')}>
                Marcar como Enviado
              </Button>
            ) : (
              <Button variant="primary" icon={<Icons.check />} iconRight={<Icons.arrowRight />} onClick={() => setStatus('aprovado')}>
                Marcar como Aprovado
              </Button>
            )}
          </div>
        </div>
      </div>

      {/* TOGGLE SINAL (temporário para testar os dois estados) */}
      <div style={{ maxWidth: 820, margin: '16px auto 0', display: 'flex', alignItems: 'center', gap: 10 }}>
        <span style={{ fontSize: 13, color: '#A29E96' }}>Sinal:</span>
        <button onClick={() => setSinal(s => !s)} style={{
          height: 30, padding: '0 12px', borderRadius: 999,
          border: `1.5px solid ${sinal ? '#2A9D8F' : '#EFEDE8'}`,
          background: sinal ? 'rgba(42,157,143,0.1)' : '#fff',
          color: sinal ? '#2A9D8F' : '#5C594F',
          fontSize: 13, fontWeight: 600, fontFamily: 'inherit', cursor: 'pointer',
        }}>
          {sinal ? 'Com sinal ✓' : 'Sem sinal'}
        </button>
      </div>

      {/* DOCUMENTO */}
      <div className="doc-scroll" style={{ margin: '16px -40px -56px', padding: '32px 20px 48px' }}>
        <div className="doc-wrap">
          <DocumentoPDF sinal={sinal} />
        </div>
      </div>

    </AppLayout>
  )
}
```
