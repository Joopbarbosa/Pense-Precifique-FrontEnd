### Componente Metric

```tsx
interface MetricProps {
  icon: React.ReactNode
  iconBg: string
  iconColor: string
  label: string
  value: string
  valueColor: string
  delta?: React.ReactNode
  deltaColor?: string
  emptyText?: string  // texto quando não há dados
}

function Metric({ icon, iconBg, iconColor, label, value, valueColor, delta, deltaColor, emptyText }: MetricProps) {
  return (
    <Card padding="20px 22px" style={{ animation: 'fadeUp .45s ease both' }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start', gap: 12 }}>
        <span style={{ fontSize: 13.5, fontWeight: 500, color: '#A29E96' }}>{label}</span>
        <span style={{
          flexShrink: 0, display: 'grid', placeItems: 'center',
          width: 42, height: 42, borderRadius: 12,
          background: iconBg, color: iconColor,
        }}>{icon}</span>
      </div>
      <div style={{ marginTop: 10, fontSize: 30, fontWeight: 700, letterSpacing: '-0.02em', color: valueColor, lineHeight: 1.1 }}>
        {value}
      </div>
      {delta ? (
        <div style={{ marginTop: 9, fontSize: 12.5, fontWeight: 600, color: deltaColor, display: 'flex', alignItems: 'center', gap: 5 }}>
          {delta}
        </div>
      ) : (
        <div style={{ marginTop: 9, fontSize: 12.5, color: '#B7B4AD' }}>{emptyText ?? '—'}</div>
      )}
    </Card>
  )
}
```

### Estrutura da página

```tsx
export default function DashboardPage() {
  const empty = false // trocar para true para ver estado vazio

  return (
    <AppLayout active="dashboard">

      {/* HEADER */}
      <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: 20, flexWrap: 'wrap', marginBottom: 26 }}>
        <div>
          <h1 style={{ margin: 0, fontSize: 30, fontWeight: 700, letterSpacing: '-0.025em', color: '#3A372F' }}>
            Dashboard
          </h1>
          <p style={{ margin: '7px 0 0', fontSize: 15, color: '#A29E96', lineHeight: 1.5 }}>
            {empty
              ? <>Bem-vinda, <strong style={{ fontWeight: 600, color: '#6B6860' }}>Ana</strong>! Vamos preparar seu negócio? Comece pelos insumos.</>
              : <>Bem-vinda de volta, <strong style={{ fontWeight: 600, color: '#6B6860' }}>Ana</strong>! Aqui está o resumo do seu negócio.</>
            }
          </p>
        </div>
        <Button
          variant="primary"
          iconRight={<Icons.arrowRight />}
          onClick={() => {}}
        >
          {empty ? 'Comece cadastrando seus insumos' : 'Novo Orçamento'}
        </Button>
      </div>

      {/* MÉTRICAS */}
      <div className="metrics">
        <Metric
          icon={<Icons.dollar />}
          iconBg="rgba(249,115,22,0.12)" iconColor="#F97316"
          label="Faturamento Mensal"
          value={empty ? 'R$ 0,00' : 'R$ 1.890,00'} valueColor="#F97316"
          delta={empty ? undefined : <><span style={{ fontSize: 13 }}>↑</span> +23% este mês</>}
          deltaColor="#3E9D5A"
          emptyText="Sem vendas ainda"
        />
        <Metric
          icon={<Icons.cube />}
          iconBg="rgba(42,157,143,0.12)" iconColor="#2A9D8F"
          label="Produtos Cadastrados"
          value={empty ? '0' : '18'} valueColor="#2A9D8F"
          delta={empty ? undefined : <><span style={{ fontSize: 13 }}>↑</span> +2 este mês</>}
          deltaColor="#3E9D5A"
          emptyText="Nenhum produto"
        />
        <Metric
          icon={<Icons.fileStack />}
          iconBg="#F1F0EC" iconColor="#7C786F"
          label="Orçamentos Pendentes"
          value={empty ? '0' : '4'} valueColor="#3A372F"
          delta={empty ? undefined : <span style={{ color: '#A29E96', fontWeight: 500 }}>Aguardando resposta</span>}
          emptyText="Nenhum pendente"
        />
      </div>

      {/* ALERTA ESTOQUE BAIXO (só com dados) */}
      {!empty && (
        <Card style={{ marginTop: 18, padding: '18px 22px', borderLeft: '4px solid #F97316', animation: 'fadeUp .5s ease both' }}>
          <div style={{ display: 'flex', alignItems: 'flex-start', gap: 14, flexWrap: 'wrap' }}>
            <span style={{
              flexShrink: 0, display: 'grid', placeItems: 'center',
              width: 40, height: 40, borderRadius: 11,
              background: 'rgba(249,115,22,0.12)', color: '#F97316', marginTop: 1,
            }}>
              <Icons.alertCircle width={20} height={20} />
            </span>
            <div style={{ flex: 1, minWidth: 220 }}>
              <div style={{ fontSize: 15.5, fontWeight: 600, color: '#3A372F' }}>3 insumos com estoque baixo</div>
              <div style={{ display: 'flex', flexWrap: 'wrap', gap: 8, marginTop: 11 }}>
                {DADOS.estoqueBaixo.map(({ nome, qtd }, i) => (
                  <span key={i} style={{
                    display: 'inline-flex', alignItems: 'center', gap: 7,
                    fontSize: 13, color: '#6B6860', whiteSpace: 'nowrap',
                    background: '#FCFBF9', border: '1px solid #EFEDE8',
                    borderRadius: 999, padding: '6px 12px',
                  }}>
                    <span style={{ width: 6, height: 6, borderRadius: '50%', background: '#F97316' }} />
                    {nome} <strong style={{ fontWeight: 600, color: '#A35A26' }}>· {qtd}</strong>
                  </span>
                ))}
              </div>
            </div>
            <a href="/insumos" style={{
              flexShrink: 0, alignSelf: 'center',
              display: 'inline-flex', alignItems: 'center', gap: 6,
              fontSize: 13.5, fontWeight: 600, color: '#F97316', textDecoration: 'none',
            }}
              onMouseOver={(e) => e.currentTarget.style.textDecoration = 'underline'}
              onMouseOut={(e) => e.currentTarget.style.textDecoration = 'none'}
            >
              Ver todos os insumos <Icons.arrowRight />
            </a>
          </div>
        </Card>
      )}

      {/* AÇÕES RÁPIDAS + DICA DO DIA */}
      <div className="lower-grid" style={{ marginTop: 18 }}>

        {/* Ações Rápidas */}
        <Card padding="22px 24px" style={{ animation: 'fadeUp .55s ease both' }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 13, marginBottom: 16 }}>
            <span style={{
              display: 'grid', placeItems: 'center', width: 42, height: 42, borderRadius: 12,
              background: 'rgba(249,115,22,0.12)', color: '#F97316',
            }}>
              <Icons.trendUp />
            </span>
            <div>
              <div style={{ fontSize: 16, fontWeight: 700, color: '#3A372F' }}>Ações Rápidas</div>
              <div style={{ fontSize: 13, color: '#A29E96', marginTop: 1 }}>
                {empty ? 'Comece por aqui' : 'Acesso direto ao essencial'}
              </div>
            </div>
          </div>
          <div style={{ display: 'flex', flexDirection: 'column', gap: 9 }}>
            {ACOES_RAPIDAS.map((acao, i) => {
              const highlight = empty && i === 0
              const IconComp = Icons[acao.icon as keyof typeof Icons]
              return (
                <a key={i} href={acao.href} style={{
                  display: 'flex', alignItems: 'center', gap: 13, padding: '13px 15px',
                  borderRadius: 12, textDecoration: 'none',
                  border: `1px solid ${highlight ? 'rgba(249,115,22,0.4)' : '#EFEDE8'}`,
                  background: highlight ? 'rgba(249,115,22,0.08)' : '#FCFBF9',
                  transition: 'background .14s, border-color .14s, transform .12s',
                }}
                  onMouseEnter={(e) => { e.currentTarget.style.background = highlight ? 'rgba(249,115,22,0.16)' : '#F6F4F0'; e.currentTarget.style.transform = 'translateX(2px)' }}
                  onMouseLeave={(e) => { e.currentTarget.style.background = highlight ? 'rgba(249,115,22,0.08)' : '#FCFBF9'; e.currentTarget.style.transform = 'none' }}
                >
                  <span style={{
                    display: 'grid', placeItems: 'center', width: 34, height: 34, borderRadius: 9,
                    background: highlight ? '#fff' : 'rgba(42,157,143,0.10)',
                    color: highlight ? '#F97316' : '#2A9D8F',
                  }}>
                    <IconComp />
                  </span>
                  <span style={{ flex: 1, fontSize: 14.5, fontWeight: 600, color: '#3A372F' }}>{acao.label}</span>
                  <span style={{ color: highlight ? '#F97316' : '#B7B4AD' }}>
                    <Icons.arrowRight />
                  </span>
                </a>
              )
            })}
          </div>
        </Card>

        {/* Dica do Dia */}
        <Card style={{
          position: 'relative', overflow: 'hidden', padding: '24px 26px',
          animation: 'fadeUp .6s ease both',
          background: 'linear-gradient(150deg, rgba(42,157,143,0.10) 0%, #FFFFFF 52%, rgba(249,115,22,0.07) 100%)',
          border: '1px solid rgba(42,157,143,0.18)',
        }}>
          {/* Blob decorativo */}
          <div style={{
            position: 'absolute', width: 150, height: 150, pointerEvents: 'none',
            borderRadius: '46% 54% 60% 40% / 50% 44% 56% 50%',
            background: 'rgba(42,157,143,0.10)', top: -60, right: -40,
            animation: 'floaty 10s ease-in-out infinite',
          }} />

          <div style={{ position: 'relative', display: 'flex', alignItems: 'center', gap: 13, marginBottom: 14 }}>
            <span style={{
              display: 'grid', placeItems: 'center', width: 46, height: 46, borderRadius: 13,
              background: '#fff', color: '#2A9D8F',
              boxShadow: '0 4px 12px -4px rgba(31,122,111,0.4)',
              border: '1px solid rgba(42,157,143,0.2)',
            }}>
              <Icons.bulb width={20} height={20} />
            </span>
            <div>
              <div style={{ fontSize: 16, fontWeight: 700, color: '#3A372F' }}>Dica do Dia</div>
              <div style={{ fontSize: 13, color: '#2A9D8F', fontWeight: 600, marginTop: 1 }}>Melhore seus lucros</div>
            </div>
          </div>
          <p style={{ position: 'relative', margin: 0, fontSize: 14.5, lineHeight: 1.62, color: '#5C594F', maxWidth: 440 }}>
            {empty
              ? 'Cadastre seus insumos com os preços reais de compra. É a base de toda precificação justa — e leva só alguns minutos.'
              : DADOS.dicaDia
            }
          </p>
        </Card>

      </div>
    </AppLayout>
  )
}
```
